# Next.js — Cheatsheet (App Router, React 19)

*Next.js is a React framework for production-grade applications.
Full-stack: server-side rendering, static generation, API routes, and more.*

### Dynamic Routes

```tsx
// src/app/blog/[slug]/page.tsx — /blog/hello-world
interface Props {
  params: Promise<{ slug: string }>;       // In Next.js 15+, params/searchParams are Promises
  searchParams: Promise<{ [key: string]: string | string[] | undefined }>;
}

export default async function BlogPost({ params }: Props) {
  const { slug } = await params;           // Need await in Next.js 15+
  const post = await getPost(slug);
  return <article>{post.content}</article>;
}

// Catch-all routes
// src/app/docs/[...slug]/page.tsx — /docs/a/b/c
export default function Docs({ params }: { params: { slug: string[] } }) {
  // params.slug = ['a', 'b', 'c']
  return <div>Docs: {params.slug.join('/')}</div>;
}

// Optional catch-all
// src/app/docs/[[...slug]]/page.tsx — /docs, /docs/a, /docs/a/b
```

    </ul>
  );

  const [count, setCount] = useState(0);
  return <button onClick={() => setCount(c => c + 1)}>{count}</button>;

## Data Fetching (Server Components)

```tsx
// Server Components can use async/await directly — no useEffect needed!

// Basic fetch (auto-cached by default)
async function getData() {
  const res = await fetch('https://api.example.com/data');
  if (!res.ok) throw new Error('Failed to fetch');
  return res.json();
}

export default async function Page() {
  const data = await getData();
  return <div>{data.map(...)}</div>;
}

// Revalidation
// Time-based (ISR)
fetch('https://api.example.com/data', { next: { revalidate: 60 } });

// On-demand revalidation
// import { revalidatePath, revalidateTag } from 'next/cache';
// revalidatePath('/blog');
// revalidateTag('posts');
// fetch(url, { next: { tags: ['posts'] } });

// Request memoization — same fetch in a tree only called once

// Parallel data fetching
export default async function Page() {
  const [dataA, dataB] = await Promise.all([
    getDataA(),
    getDataB(),
  ]);
  return <div>...</div>;
}

// Sequential data fetching (when one depends on other)
async function getArtist(id: string) {
  const artist = await db.artists.findUnique({ where: { id } });
  const albums = await db.albums.findMany({ where: { artistId: id } });
  return { artist, albums };
}
```

## Route Handlers (API)

```tsx
// src/app/api/users/route.ts
import { NextRequest, NextResponse } from 'next/server';

// GET
export async function GET(request: NextRequest) {
  const searchParams = request.nextUrl.searchParams;
  const page = searchParams.get('page') ?? '1';

  const users = await db.users.findMany({ skip: (Number(page) - 1) * 10, take: 10 });
  return NextResponse.json(users);
}

// POST
export async function POST(request: NextRequest) {
  const body = await request.json();
  const user = await db.users.create({ data: body });
  return NextResponse.json(user, { status: 201 });
}

// PUT, PATCH, DELETE
export async function PUT(request: NextRequest) { ... }
export async function DELETE(request: NextRequest) { ... }

// Dynamic route handlers
// src/app/api/users/[id]/route.ts
export async function GET(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  const user = await db.users.findUnique({ where: { id } });
  if (!user) return NextResponse.json({ error: 'Not found' }, { status: 404 });
  return NextResponse.json(user);
}

// Cached responses (force-cache or revalidate)
export const dynamic = 'force-static';   // Static at build time
// export const revalidate = 3600;        // ISR every hour

// Redirects
export async function GET() {
  return NextResponse.redirect(new URL('/login', request.url));
  // return NextResponse.rewrite(new URL('/maintenance', request.url));  // Rewrite (URL stays)
}

// Response types
return NextResponse.json({ data });
return new Response('Plain text', { status: 200 });
return NextResponse.redirect(url);
return NextResponse.json(data, { status: 201, headers: { 'X-Custom': 'value' } });
```

## Middleware

```tsx
// src/middleware.ts (runs on every request at the edge)
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // Authentication check
  const token = request.cookies.get('session')?.value;
  if (!token && pathname.startsWith('/dashboard')) {
    return NextResponse.redirect(new URL('/login', request.url));
  }

  // Add headers
  const response = NextResponse.next();
  response.headers.set('x-custom-header', 'hello');
  return response;
}

// Matcher config — which paths the middleware runs on
export const config = {
  matcher: [
    '/((?!api|_next/static|_next/image|favicon.ico).*)',  // Exclude static files
    '/dashboard/:path*',
  ],
};

// Using cookies
// Read:
request.cookies.get('name');
// Set:
response.cookies.set('name', 'value', { httpOnly: true, secure: true, maxAge: 3600 });
response.cookies.delete('name');

// Using geolocation / UA (with Vercel)
// request.geo?.city
// request.geo?.country
// request.ua?.ua
```

## Rendering Strategies

```tsx
// === STATIC RENDERING (default for fetch without revalidate) ===
// Pre-rendered at build time, serves cached HTML
export default async function StaticPage() { ... }

// === DYNAMIC RENDERING ===
// Rendered per request (use when: cookies, headers, searchParams, or dynamic functions)
export default async function DynamicPage() {
  const data = await fetch('https://api.com/data', { cache: 'no-store' });
  // OR:
  // export const dynamic = 'force-dynamic';
  // export const revalidate = 0;
}

// === ISR (Incremental Static Regeneration) ===
// Static with periodic revalidation
export default async function ISRPage() {
  const data = await fetch('https://api.com/data', { next: { revalidate: 60 } });
  // OR:
  // export const revalidate = 60;  // Revalidate every 60s
}

// === Edge Runtime ===
// For Route Handlers or Middleware
// export const runtime = 'edge';  // Runs on Edge (fast, limited Node APIs)
// Default: 'nodejs'

// Route Segment Config
export const dynamic = 'auto';             // 'auto' | 'force-dynamic' | 'force-static' | 'error'
export const dynamicParams = true;          // true | false
export const revalidate = false;            // false | number (seconds)
export const fetchCache = 'auto';           // 'auto' | 'default-cache' | 'only-cache' | 'force-cache' | 'force-no-store'
export const runtime = 'nodejs';            // 'nodejs' | 'edge'
export const preferredRegion = 'auto';      // 'auto' | 'home' | 'iad1' | specific Vercel region
```

## Configuration

```ts
// next.config.ts (or .mjs)
import type { NextConfig } from 'next';

const nextConfig: NextConfig = {
  // Environment variables (public — inlined at build time)
  env: {
    CUSTOM_KEY: 'value',
  },

  // Build-time env (exposed to client)
  publicRuntimeConfig: {
    API_URL: process.env.NEXT_PUBLIC_API_URL,
  },

  // Redirects (runs before middleware)
  async redirects() {
    return [
      { source: '/old-page', destination: '/new-page', permanent: true },
      { source: '/post/:slug', destination: '/blog/:slug', permanent: true },
    ];
  },

  // Rewrites (URL mapping without redirect)
  async rewrites() {
    return [
      { source: '/api/:path*', destination: 'https://api.example.com/:path*' },  // Proxy
    ];
  },

  // Headers
  async headers() {
    return [
      {
        source: '/(.*)',
        headers: [
          { key: 'X-Frame-Options', value: 'DENY' },
          { key: 'X-Content-Type-Options', value: 'nosniff' },
          { key: 'Referrer-Policy', value: 'strict-origin-when-cross-origin' },
        ],
      },
      {
        source: '/fonts/(.*)',
        headers: [
          { key: 'Cache-Control', value: 'public, max-age=31536000, immutable' },
        ],
      },
    ];
  },

  // Output (standalone for Docker)
  output: 'standalone',
};

export default nextConfig;
```
